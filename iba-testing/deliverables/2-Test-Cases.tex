\section{Introduction}

\subsection{Purpose}
This document defines the formal Test Case Specification for the IBA Campus Facility Booking System (CFBS). It details the inputs, procedural execution steps, expected outcomes, and actual execution statuses observed during the automated and manual test runs.

\subsection{Scope & Traceability}
These test cases are designed to validate all functional requirements documented in the CFBS Software Requirements Specification (SRS). They span multiple test levels:
\begin{itemize}[noitemsep]
    \item \textbf{Unit Level (NestJS / Jest):} Validating individual services, helper utilities, and controllers in isolation.
    \item \textbf{Integration Level (Supertest / Express):} Testing REST API endpoint routing, middleware authorization, and database persistence.
    \item \textbf{E2E System Level (Playwright / Chromium \& Firefox):} Emulating complete user workflows in headless and headed browser contexts.
\end{itemize}

\subsection{Environmental & Procedural Requirements}
To execute the test cases specified herein, the following sandbox parameters are strictly required:
\begin{enumerate}[noitemsep]
    \item A dedicated, isolated testing database seeded via \texttt{seedTestData.sql} before each test suite execution.
    \item A local test server listening on port \texttt{3000} (Backend API) and a Vite development server on port \texttt{5173} (Frontend Client).
    \item Automation frameworks configured using Node 18+, Jest v30, and Playwright v1.60.
\end{enumerate}

\section{Test Case Specifications}

\subsection{Authentication & Authorization (TC-AUTH)}

\ieeetestcase{TC-AUTH-001}{Valid Student Login Validation}{E2E \& Integration}{Positive / Functional}{REQ-1, REQ-3, REQ-4}{\textbf{PASS}}{Database is seeded with user \texttt{test-student} and password \texttt{testpass}.}{
    \item Send a POST request to \texttt{/api/auth/login} with student credentials or enter them on the login page.
    \item Verify that the response returns an \texttt{access\_token} and a payload containing the student's profile.
    \item Confirm that the student is redirected to \texttt{/dashboard} and that the plain text password is not returned.
}

\ieeetestcase{TC-AUTH-002}{Valid Program Office Login Validation}{E2E \& Integration}{Positive / Functional}{REQ-2, REQ-3, REQ-4}{\textbf{PASS}}{Database is seeded with PO user \texttt{test-po} and password \texttt{testpass}.}{
    \item Submit a POST request to \texttt{/api/auth/login} with PO credentials.
    \item Assert that a valid JWT token is returned and that the role is identified as \texttt{programoffice}.
    \item Verify that the user is redirected to the Program Office dashboard.
}

\ieeetestcase{TC-AUTH-003}{Invalid Credentials Handling}{E2E, Integration, Unit}{Negative / Security}{REQ-5}{\textbf{PASS}}{The auth database is active and populated.}{
    \item Send a login request using an invalid ERP or password (e.g., username: \texttt{wrong-user}, password: \texttt{wrong-pass}).
    \item Assert that the API rejects the request with a \texttt{401 Unauthorized} status.
    \item Verify that the UI displays the error message: ``Invalid credentials''.
}

\ieeetestcase{TC-AUTH-004}{Protected Route Access Redirection}{Integration (RBAC)}{Security / Negative}{REQ-6}{\textbf{PASS}}{The user is unauthenticated (no session token stored).}{
    \item Attempt to directly access a protected URL (e.g., \texttt{GET /api/bookings}).
    \item Verify that the API rejects the request with a \texttt{401 Unauthorized} status.
    \item Confirm that the UI redirects the unauthenticated user back to the login page.
}

\subsection{Room Booking Request (TC-BOOK)}

\ieeetestcase{TC-BOOK-001}{Happy Path Booking Creation}{E2E \& Integration}{Positive / Functional}{REQ-1, REQ-2, REQ-3, REQ-6}{\textbf{PASS} (Firefox Cleanup Warning)}{Database is seeded; user \texttt{test-student} is authenticated.}{
    \item Select an active building (Test Building) and room (Test Room A) from the dropdown options.
    \item Choose a date in the current year and select an available time slot (Slot 3: 11:30 AM).
    \item Enter a valid booking purpose and submit the request.
    \item Assert that the request is created with a status of \texttt{pending} and appears in the student's dashboard.
}

\ieeetestcase{TC-BOOK-002}{Overlapping Booking Prevention}{E2E \& Integration}{Negative / Constraint}{REQ-4, REQ-7}{\textbf{PASS}}{A pending booking exists for Room A on a specific date and time slot.}{
    \item Authenticate as a student.
    \item Attempt to submit a new booking request for the exact same room, date, and slot.
    \item Verify that the system rejects the request with a \texttt{409 Conflict} status and does not submit it.
}

\ieeetestcase{TC-BOOK-003}{Blocked Slot Booking Prevention}{E2E \& Integration}{Negative / Constraint}{REQ-4, REQ-7}{\textbf{PASS}}{An Admin has blocked Slot 2 on the target date.}{
    \item Authenticate as a student.
    \item Attempt to book Slot 2 on the blocked date for the specified room.
    \item Assert that the system rejects the request with a \texttt{409 Conflict} status, returning an error: ``This slot is blocked by admin''.
}

\ieeetestcase{TC-BOOK-004/5/6/7}{Booking Boundary Verification}{Integration}{Boundary}{REQ-3}{\textbf{PASS}}{The database is active; student is authenticated.}{
    \item Send a POST request to \texttt{/api/bookings} with an invalid \texttt{slot\_id} (e.g., lower bound: \texttt{0}, upper bound: \texttt{8}).
    \item Assert that the API rejects the request with a \texttt{400 Bad Request} validation error.
    \item \textbf{Result:} The system correctly returns a \texttt{400 Bad Request} validation error for out-of-bounds slot IDs.
}

\ieeetestcase{TC-BOOK-008}{Missing Purpose Field Validation}{E2E \& Integration}{Negative / Validation}{REQ-5}{\textbf{PASS}}{The student is logged in and preparing a booking form.}{
    \item Select a building, room, date, and slot, but leave the booking ``purpose'' field empty.
    \item Click ``Submit Booking Request.''
    \item Verify that native HTML5 validations or custom UI alert boxes block the submission and highlight the missing field.
}

\ieeetestcase{TC-BOOK-009}{Simultaneous Booking Concurrency}{Integration}{Concurrency / Stress}{REQ-7}{\textbf{PASS}}{Two students are logged in simultaneously.}{
    \item Fire two booking requests for the exact same room, date, and slot at the exact same millisecond.
    \item Assert that one request succeeds (\texttt{201 Created}) while the other is gracefully rejected with a \texttt{409 Conflict}.
    \item \textbf{Result:} One request succeeds and the concurrent duplicate is correctly rejected with \texttt{409 Conflict}.
}

\subsection{Booking Management --- Program Office (TC-PO)}

\ieeetestcase{TC-PO-001}{View Pending Requests Dashboard}{E2E \& Integration}{Positive / Functional}{REQ-1}{\textbf{PASS}}{A student has submitted a pending booking request.}{
    \item Log in to the portal using Program Office credentials (\texttt{test-po}).
    \item Access the pending requests tab.
    \item Assert that the queue correctly displays the student's name, room, date, slot, and purpose.
}

\ieeetestcase{TC-PO-002}{Approve Pending Request}{E2E \& Integration}{Positive / Functional}{REQ-2, REQ-3, REQ-4}{\textbf{PASS}}{A pending request exists in the Program Office queue.}{
    \item Log in as a PO member and select the target pending request.
    \item Click ``Approve.''
    \item Verify that the booking status is updated to \texttt{approved} and moves to the approved bookings tab.
}

\ieeetestcase{TC-PO-003}{Reject Pending Request}{E2E \& Integration}{Positive / Functional}{REQ-2, REQ-4}{\textbf{PASS}}{A pending request exists in the Program Office queue.}{
    \item Log in as a PO member and select the target pending request.
    \item Click ``Reject.''
    \item Assert that the booking status changes to \texttt{rejected} and the time slot remains available for other users.
}

\ieeetestcase{TC-PO-004}{Auto-Rejection of Overlapping Bookings}{E2E \& Integration}{Functional / Logic}{REQ-5}{\textbf{PASS}}{Two pending booking requests exist for the same room and slot.}{
    \item Log in as a PO member.
    \item Approve the first student's pending request.
    \item Verify that the second student's request is automatically updated to \texttt{rejected} or flagged as a conflict.
    \item \textbf{Result:} Both requests appear in the PO queue. Upon approving the first, the system automatically rejects the conflicting second request.
}

\subsection{Booking Cancellation (TC-CANCEL)}

\ieeetestcase{TC-CANCEL-001}{Student Cancels Own Pending Booking}{E2E \& Integration}{Positive / Functional}{REQ-1, REQ-2, REQ-3}{\textbf{PASS}}{A student has a booking request with a \texttt{pending} status.}{
    \item Log in as the student who created the booking and locate it on the dashboard.
    \item Click ``Cancel'' and accept the confirmation dialog.
    \item Assert that the booking status changes to \texttt{cancelled}.
    \item \textbf{Result:} The booking status is correctly updated to \texttt{cancelled} in the database.
}

\ieeetestcase{TC-CANCEL-002}{Student Cancels Own Approved Booking}{E2E}{Positive / Functional}{REQ-1, REQ-2, REQ-3}{\textbf{PASS}}{A student has a booking request with an \texttt{approved} status.}{
    \item Log in as the student who created the booking and view the dashboard.
    \item Locate the card for the approved booking.
    \item Click ``Cancel'' and accept the confirmation dialog.
    \item \textbf{Result:} The ``Cancel'' button is visible for approved bookings and cancellation completes successfully.
}

\ieeetestcase{TC-CANCEL-005}{PO Cancels Approved Booking}{E2E \& Integration}{Positive / Functional}{REQ-1, REQ-2, REQ-3}{\textbf{PASS}}{An approved booking exists in the system.}{
    \item Log in as a PO member and access the approved bookings tab.
    \item Locate the target booking, click ``Cancel,'' and accept the confirmation dialog.
    \item Assert that the booking status changes to \texttt{cancelled}.
    \item \textbf{Result:} The booking status is correctly updated to \texttt{cancelled} in both the UI and the database.
}

\subsection{Admin System Management & Data Integrity (TC-ADMIN / TC-DATA)}

\ieeetestcase{TC-ADMIN-001}{Create Student Account}{E2E \& Integration}{Positive / Functional}{REQ-1, REQ-3}{\textbf{PASS}}{The Admin is logged in.}{
    \item Access the student management tab.
    \item Enter a unique ERP, name, email, and password, and submit the form.
    \item Verify that the student record is created in the database and visible in the UI.
}

\ieeetestcase{TC-ADMIN-002}{Create Duplicate Student Account}{E2E \& Integration}{Negative / Functional}{REQ-2, REQ-4}{\textbf{PASS}}{A student record with ERP \texttt{12345} already exists.}{
    \item Attempt to create a new student account using the existing ERP \texttt{12345}.
    \item Assert that the system rejects the creation with a \texttt{409 Conflict} status and displays a unique constraint error.
}

\ieeetestcase{TC-ADMIN-006/007}{Block & Unblock Slot Lifecycle}{Integration}{Positive / Functional}{REQ-1, REQ-2, REQ-3}{\textbf{PASS}}{The Admin is logged in.}{
    \item Send a POST request to \texttt{/api/blocked-slots} to block Slot 1 on a specific room and date.
    \item Verify that the block is recorded in the database.
    \item Send a DELETE request to \texttt{/api/blocked-slots/:id} to unblock the slot.
    \item Assert that the system returns a \texttt{200 OK} status with the message: ``Slot unblocked''.
}

\ieeetestcase{TC-DATA-001}{Audit Trail Verification}{Integration}{Functional / Security}{REQ-8}{\textbf{PASS}}{A booking has been approved by the Program Office.}{
    \item Retrieve the approved booking record by sending a GET request to \texttt{/api/bookings/:id}.
    \item Assert that the returned payload contains the \texttt{reviewed\_by} field containing the PO member's UUID.
    \item \textbf{Result:} The returned payload correctly includes the \texttt{reviewed\_by} field with the PO member's UUID, confirming audit trail integrity.
}
